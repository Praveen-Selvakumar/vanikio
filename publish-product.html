// Wait until QRCode.js creates the canvas, // then apply the logo. setTimeout(() => { applyLogo(); }, 50); } // ============================================================ // APPLY LOGO TO CENTER OF QR // ============================================================ function applyLogo() { const qr = getElement("qrcode"); }); this.classList.add( "active" ); generateQr(); } ); }); } // ============================================================ // INPUT AUTO UPDATE // ============================================================ function initializeInputListeners() { const inputIds = [ "businessId", "productName", "productDescription", "serial", "price", "itemId" ]; Please try again."; } if (error.message) { return error.message; } return "Unable to publish product."; } // ============================================================ // DIALOGS // ============================================================ function showLoadingDialog() { const dialog = getElement("loadingDialog"); if (!businessId) { return "Please enter Business ID."; } if (!productName) { return "Please enter Product Name."; } if (!productDescription) { return "Please enter Product Description."; } if (!itemId) { return "Please enter QR / Item ID."; } return ""; } // ============================================================ // DOWNLOAD QR PNG // ============================================================ async function downloadQrPng() { const qrCard = getElement("qrCard"); const itemId = getElement("itemId")?.value.trim() || "QR"; const link = document.createElement("a"); link.download = "VANIKIO-QR-" + itemId + ".png"; link.href = canvas.toDataURL("image/png"); link.click(); } // ============================================================ // PUBLISH PRODUCT // ============================================================ async function publishProduct() { const validationError = validateProduct(); if (validationError) { showErrorDialog( validationError ); return; } showLoadingDialog(); try { const businessId = getElement("businessId") .value .trim(); const itemId = getElement("itemId") .value .trim(); const productName = getElement("productName") .value .trim(); const productDescription = getElement("productDescription") .value .trim(); const serial = getElement("serial") .value .trim(); const price = getElement("price") .value .trim(); const qrUrl = "https://vanikio.com/i/" + encodeURIComponent( businessId ) + "/" + encodeURIComponent( itemId ); const productRef = doc( db, "businesses", businessId, "products", itemId ); await setDoc( productRef, { businessId: businessId, itemId: itemId, productName: productName, productDescription: productDescription, serial: serial, price: price, qrEnabled: true, qrUrl: qrUrl, borderColor: selectedColors.border, textColor: selectedColors.text, hasLogo: !!selectedLogo, updatedAt: serverTimestamp() }, { merge: true } ); // ==================================================== // DOWNLOAD PNG AFTER SUCCESSFUL PUBLISH // ==================================================== await downloadQrPng(); hideLoadingDialog(); showSuccessDialog(); } catch (error) { console.error( "Publish product failed:", error ); hideLoadingDialog(); showErrorDialog( getErrorMessage(error) ); } } // ============================================================ // RESET QR // ============================================================ function resetQr() { Object.entries(DEFAULTS) .forEach(([key, value]) => { const element = getElement(key); if (!element) { return; } element.value = value; }); selectedColors = { ...DEFAULT_COLORS }; selectedLogo = null; const logoInput = getElement("logoInput"); if (logoInput) { logoInput.value = ""; } document .querySelectorAll(".color-option") .forEach(option => { option.classList.remove( "active" ); }); document .querySelectorAll(".color-option") .forEach(option => { const target = option.dataset.target; const color = option.dataset.color; if ( DEFAULT_COLORS[target] === color ) { option.classList.add( "active" ); } }); generateQr(); } // ============================================================ // LANGUAGE // ============================================================ function setLanguage(language) { if ( language !== "en" && language !== "ta" ) { language = "en"; } localStorage.setItem( "vanikioProductLanguage", language ); document.documentElement.lang = language; if ( typeof translations !== "undefined" && translations[language] ) { document .querySelectorAll( "[data-i18n]" ) .forEach(element => { const key = element.dataset.i18n; if ( translations[language][key] ) { element.textContent = translations[language][key]; } }); } document .querySelectorAll( "[data-placeholder-en]" ) .forEach(element => { const placeholder = language === "ta" ? element.dataset.placeholderTa : element.dataset.placeholderEn; if (placeholder) { element.placeholder = placeholder; } }); const englishButton = getElement("englishButton"); const tamilButton = getElement("tamilButton"); if (englishButton) { englishButton.classList.toggle( "active", language === "en" ); } if (tamilButton) { tamilButton.classList.toggle( "active", language === "ta" ); } } // ============================================================ // INITIALIZE PAGE // ============================================================ function initializePage() { loadSavedBusinessId(); initializeColorPalette(); initializeInputListeners(); initializeLogoInput(); const savedLanguage = localStorage.getItem( "vanikioProductLanguage" ) || "en"; setLanguage( savedLanguage ); generateQr(); } // ============================================================ // GLOBAL FUNCTIONS // ============================================================ window.publishProduct = publishProduct; window.resetQr = resetQr; window.setLanguage = setLanguage; window.closeSuccessDialog = closeSuccessDialog; window.closeErrorDialog = closeErrorDialog; // ============================================================ // START // ============================================================ if ( document.readyState === "loading" ) { document.addEventListener( "DOMContentLoaded", initializePage ); } else { initializePage(); } ``` ### `publish-product.css` ```css /* ========================================================= THEME ========================================================= */ :root { --bg: #f7f5ff;
